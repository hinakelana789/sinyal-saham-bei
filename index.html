function parseWatchlist() { const raw = $("rankSymbols").value.trim(); if (!raw) { throw new Error("Masukkan minimal satu kode saham."); } const symbols = [...new Set( raw.toUpperCase() .split(/[\s,;]+/) .map(s => s.trim()) .filter(Boolean) )]; if (symbols.length > 10) { throw new Error("Maksimal 10 kode saham dalam satu pemindaian."); } const invalid = symbols.filter(s => !/^[A-Z0-9]{3,5}$/.test(s)); if (invalid.length) { throw new Error("Kode saham tidak valid: " + invalid.join(", ")); } return symbols; } async function fetchRows(symbol) { const url = "/.netlify/functions/stock-history?symbol=" + encodeURIComponent(symbol); const response = await fetch(url, { headers: {Accept: "application/json"} }); let json; try { json = await response.json(); } catch { throw new Error("Respons server bukan JSON yang valid."); } if (!response.ok) { throw new Error( json?.error || json?.message || "Server gagal mengambil data (" + response.status + ")." ); } const rawRows = extractRows(json); const rows = sortRows(rawRows); if (!rows.length) { throw new Error("Data harga kosong untuk " + symbol + "."); } return rows; } /* * Cari tanggal terakhir yang tersedia pada semua saham * yang berhasil dimuat. */ function alignToCommonDate(results) { const usable = results.filter(a => !a.error && Array.isArray(a.rows) && a.rows.length > 0 ); if (!usable.length) { return {commonDate: null, results}; } const dateSets = usable.map(a => new Set(a.rows.map(row => row.date)) ); const candidateDates = [...dateSets[0]] .sort((a, b) => b.localeCompare(a)); const commonDate = candidateDates.find(date => dateSets.every(set => set.has(date)) ); if (!commonDate) { return { commonDate: null, results: results.map(a => ({ ...a, alignmentExcluded: !a.error })) }; } const alignedResults = results.map(a => { if (a.error || !Array.isArray(a.rows)) return a; const rows = a.rows.filter(row => row.date <= commonDate); const last = rows[rows.length - 1]; if (!last || last.date !== commonDate) { return { ...a, alignmentExcluded: true }; } const recalculated = calcAnalysis(rows); return { ...recalculated, symbol: a.symbol, alignmentExcluded: false }; }); return {commonDate, results: alignedResults}; } function rankLabel(a) { if (a.error) return "GAGAL MEMUAT"; if (a.alignmentExcluded) return "TANGGAL BERBEDA"; if (!a.enough || a.score === null) return "DATA KURANG"; return a.signal; } async function runRanking() { const btn = $("rankBtn"); let symbols; try { symbols = parseWatchlist(); } catch (error) { setText("rankStatus", error.message); return; } btn.disabled = true; $("rankRows").innerHTML = "Mengambil data saham…"; setText("rankStatus", "Membandingkan " + symbols.length + " saham…"); try { const loaded = await Promise.all(symbols.map(async symbol => { try { const rows = await fetchRows(symbol); const analysis = calcAnalysis(rows); return {...analysis, symbol}; } catch (error) { return { symbol, error: error.message, score: null }; } })); const aligned = alignToCommonDate(loaded); const ranked = [...aligned.results].sort((a, b) => { const aGood = !a.error && !a.alignmentExcluded && a.enough && a.score !== null; const bGood = !b.error && !b.alignmentExcluded && b.enough && b.score !== null; if (aGood !== bGood) return aGood ? -1 : 1; if (aGood && bGood) return b.score - a.score; return a.symbol.localeCompare(b.symbol); }); $("rankRows").innerHTML = ranked.map((a, index) => { const valid = !a.error && !a.alignmentExcluded && a.enough && a.score !== null; return "" + "" + (valid ? index + 1 : "—") + "" + "" + esc(a.symbol) + "" + "" + rupiah(a.last?.close) + "" + "" + esc(a.last?.date || "—") + "" + "" + (valid ? fmt(a.score) : "—") + "" + "" + esc(rankLabel(a)) + "" + ""; }).join(""); const count = ranked.filter(a => !a.error && !a.alignmentExcluded && a.enough && a.score !== null ).length; setText("rankStatus", "Tanggal pembanding: " + (aligned.commonDate || "tidak tersedia") + ". " + count + " saham memenuhi syarat pemeringkatan. " + "Skor bukan probabilitas keuntungan." ); } catch (error) { setText("rankStatus", "Perbandingan gagal: " + error.message); } finally { btn.disabled = false; } } async function runRadar() { const btn = $("radarRefresh"); let symbols; try { symbols = parseWatchlist(); } catch (error) { setText("radarStatus", error.message); return; } btn.disabled = true; setText("radarStatus", "Memindai " + symbols.length + " saham…"); setText("radarUpdated", "Memuat data…"); $("radarRows").innerHTML = "Sedang memuat data…"; try { const loaded = await Promise.all(symbols.map(async symbol => { try { const rows = await fetchRows(symbol); return {...calcAnalysis(rows), symbol}; } catch (error) { return { symbol, error: error.message, score: null }; } })); const aligned = alignToCommonDate(loaded); const all = aligned.results; const good = all.filter(a => !a.error && !a.alignmentExcluded && a.last?.date === aligned.commonDate && a.enough && a.score !== null ); const sorted = [ ...good.sort((a, b) => b.score - a.score), ...all.filter(a => a.error || a.alignmentExcluded || !a.enough || a.score === null ) ]; const strong = good.filter(a => a.signal === "BELI BERSYARAT" ).length; const weak = good.filter(a => a.signal.includes("HINDARI") || a.signal.includes("RISIKO") ).length; const watch = good.length - strong - weak; setText("radarStrong", String(strong)); setText("radarWatch", String(watch)); setText("radarWeak", String(weak)); $("radarRows").innerHTML = sorted.map(a => { if (a.error) { return "" + esc(a.symbol) + "Gagal memuat: " + esc(a.error) + "—"; } if (a.alignmentExcluded) { return "" + esc(a.symbol) + "" + rupiah(a.last?.close) + "" + esc(a.last?.date || "—") + "——" + "DIKECUALIKAN — tanggal berbeda—"; } if (!a.enough || a.score === null) { return "" + esc(a.symbol) + "" + rupiah(a.last?.close) + "" + esc(a.last?.date || "—") + "" + (a.dailyChange === null ? "—" : fmt(a.dailyChange) + "%") + "Data kurang" + esc(a.signal) + "—"; } return "" + esc(a.symbol) + "" + rupiah(a.last.close) + "" + esc(a.last.date) + "" + (a.dailyChange === null ? "—" : (a.dailyChange > 0 ? "+" : "") + fmt(a.dailyChange) + "%") + "" + (a.trendPositive ? "Positif" : a.trendNegative ? "Negatif" : "Campuran") + "" + esc(a.signal) + "" + fmt(a.score) + ""; }).join(""); setText("radarStatus", "Tanggal pembanding: " + (aligned.commonDate || "tidak tersedia") + ". " + good.length + " saham memenuhi syarat pemeringkatan. " + "Data ini bukan jaminan real-time." ); setText("radarUpdated", "Pembaruan terakhir: " + new Date().toLocaleString("id-ID") ); } catch (error) { setText("radarStatus", "Radar gagal: " + error.message); setText("radarUpdated", "Pembaruan gagal."); } finally { btn.disabled = false; } }
